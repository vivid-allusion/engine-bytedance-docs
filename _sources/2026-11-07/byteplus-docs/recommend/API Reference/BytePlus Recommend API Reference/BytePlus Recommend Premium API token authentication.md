This topic describes how to create a signature and add it to an HTTP request that is sent to BytePlus Recommend for BytePlus Recommend Premium clients.

You must use token authentication in your HTTP headers to prove your identity's legitimacy and avoid any malicious interception of a request during transfer. This mainly identifies our clients and provides extra security measures during request calls.

<span id="3ec8d806"></span>
# Token parameters

These parameters are used in the HTTP header to identify and authenticate our clients.


|**Parameters** |**Data Type** |**Required** |**Sample Value** |**Description** |
|---|---|---|---|---|
|ProjectID |String |Yes |saas_retail_demo |The unique ID name assigned to BytePlus clients. |
|Token |String |Yes |85d624c00b35064babb8d4aa91e328c9 |The unique authentication and encryption key provided to each client by the BytePlus Recommend team. |
|Tenant\-Id |String |Yes |3000001813 |The client ID provided by the BytePlus Recommend team. |


<span id="a413af7d"></span>
# Request example

The following is an HTTP header example in the API request:

```JSON
{
    "ProjectID": "saas_retail_demo",
    "Token": "85d624c00b35064babb8d4aa91e328c9",
    "Tenant-Id": "3000001813"
}
```

# HTTP header parameters

Below are detailed descriptions of shared request HTTP parameters.


|**Parameter Name** |**Data Type** |**Sample Value** |**Location** |**Description** |
|---|---|---|---|---|
|Tenant\-id |String |3000001813 |Header |The BytePlus Recommend team will distribute this with the private key. The Tenant\-id is used to identify clients. |
|Tenant\-Ts |String |1645067393 |Header |The current timestamp, in seconds, will be used to verify whether the request has expired. The expiration time is limited to 5 seconds. |
|Tenant\-Nonce |String |1645067393 |Header |Generated random string. You can use the current timestamp. |
|Tenant\-Signature |String |7a73b4193a46727af2b7355757fc2e847c5659107d7a806bdecd6e46cf9644ab |Header |The client's signature verification. The following values are required:<br><br><br>* Token: The authentication and encryption key assigned to each client.<br><br>* Tenant\-Ts: The current timestamp in seconds.<br><br>* Tenant\-Nonce: Generated random string. |
|Request\-Id |String |84kduxkls74lcdj73jdu3 |Header |The request ID generated by the client. There is no specific format to adhere to as long as each ID is unique. It is recommended that the request ID not be too long. |
|Content\-Type |String |application/x\-protobuf |Header |The request body serialization method supports the following two methods:<br><br><br>* `application/x-protobuf`<br><br>* `application/json`<br><br><br>For more information, refer to the specific interface descriptions. |
|Accept |String |application/json |Header |The serialization method of the request body. If the header is not available, the `content-type` serialization will be used. |

```Python
package main

import (
    "crypto/sha256"
    "encoding/json"
    "fmt"
    "log"
    "strconv"
    "time"

    "github.com/google/uuid"
    "github.com/valyala/fasthttp"
)

//  Your credentials
const (
    projectID = "saas_retail_demo"
    token     = "85d624c00b35064babb8d4aa91e328c9"
    tenantID  = "3000001813"
)

// Compute signature
func calSignature(token, tenantID string, bodyBytes []byte, tenantTs, tenantNonce string) string {
    shaHash := sha256.New()
    shaHash.Write([]byte(token))
    shaHash.Write(bodyBytes)
    shaHash.Write([]byte(tenantID))
    shaHash.Write([]byte(tenantTs))
    shaHash.Write([]byte(tenantNonce))
    // convert to hexadecimal
    signature := fmt.Sprintf("%x", shaHash.Sum(nil))
    return signature
}

var (
    writeUsersUrl = "https://rec-ap-singapore-1.byteplusapi.com/RetailSaaS/WriteUsers"
    nowTime       = strconv.FormatInt(time.Now().Unix(), 10)
    userData      = map[string]interface{}{
       "user_id": "123",
       "gender":  "female",
    }
    userDataBytes, _ = json.Marshal(userData)
)

type Status struct {
    Code    int    `json:"code"`
    Message string `json:"message"`
}

type DataError struct {
    Message string `json:"message"`
    Data    string `json:"data"`
}

type Response struct {
    Status Status       `json:"status"`
    Errors []*DataError `json:"errors"`
}

func main() {
    writeUserRequest := map[string]interface{}{
       "project_id": projectID,
       "stage":      "production",
       "data": []string{
          string(userDataBytes),
       },
    }
    bodyBytes, _ := json.Marshal(writeUserRequest)
    headers := map[string]string{
       "Tenant-Id":        tenantID,
       "Tenant-Ts":        nowTime,
       "Tenant-Nonce":     nowTime,
       "Tenant-Signature": calSignature(token, tenantID, bodyBytes, nowTime, nowTime),
       "Request-Id":       uuid.New().String(),
       "Content-Type":     "application/json",
       "Accept":           "application/json",
    }

    request := fasthttp.AcquireRequest()
    response := fasthttp.AcquireResponse()
    defer func() {
       fasthttp.ReleaseRequest(request)
       fasthttp.ReleaseResponse(response)
    }()
    request.SetRequestURI(writeUsersUrl)
    request.Header.SetMethod(fasthttp.MethodPost)
    for key, value := range headers {
       request.Header.Set(key, value)
       fmt.Println(key + ": " + value)
    }
    request.SetBodyRaw(bodyBytes)
    client := &fasthttp.Client{}
    err := client.DoTimeout(request, response, 10*time.Second)
    if err != nil {
       log.Printf("write occurred error: %s\n", err.Error())
    }
    // Usually 200 even with write errors.
    log.Printf("response status: %d\n", response.StatusCode())
    if response.StatusCode() != 200 {
       return
    }
    resp := &Response{}
    _ = json.Unmarshal(response.Body(), resp)
    // real code, if the request is successful, this field will not be returned, so it will be '0' here.
    log.Printf("code: %d\n", resp.Status.Code)
    // If the request is successful, the message value is 'OK', otherwise it will be a failure message.
    log.Printf("Message: %s\n", resp.Status.Message)
    if resp.Status.Code != 0 {
       for _, err := range resp.Errors {
          log.Printf("failure message: %s, failure data sample: %s\n", err.Message, err.Data)
       }
    }
}
```

```Python
import hashlib
import json
import time
import uuid

import requests

# Your credentials
project_id = "saas_retail_demo"
token = "85d624c00b35064babb8d4aa91e328c9"
tenant_id = "3000001813"

# Compute signature
def cal_signature(m_token: str, m_tenant_id: str, m_body_bytes: bytes, tenant_ts: str, tenant_nonce: str) -> str:
    sha256 = hashlib.sha256()
    sha256.update(m_token.encode("utf-8"))
    sha256.update(m_body_bytes)
    sha256.update(m_tenant_id.encode("utf-8"))
    sha256.update(tenant_ts.encode("utf-8"))
    sha256.update(tenant_nonce.encode("utf-8"))

    return sha256.hexdigest()

write_users_url = "https://rec-ap-singapore-1.byteplusapi.com/RetailSaaS/WriteUsers"
now_time = str(int(time.time()))
user_data = {
    "user_id": "123",
    "gender": "female",
}
userDataBytes = json.dumps({
    "users": [
        {
            "user_id": "1",
            "age": "18"
        }
    ]
}).encode("utf-8")


if __name__ == '__main__':
    write_users_request = json.dumps({
        "project_id": project_id,
       "stage":      "production",
        "data": [
            json.dumps(user_data)
        ]
    }).encode("utf-8")
    headers = {
        "Tenant-Id": tenant_id,
        "Tenant-Ts": now_time,
        "Tenant-Nonce": now_time,
        "Tenant-Signature": cal_signature(token, tenant_id, write_users_request, now_time, now_time),
        "Request-Id": str(uuid.uuid4()),
        "Content-Type": "application/json",
        "Accept": "application/json",
    }
    response = requests.request("POST", url=write_users_url, data=write_users_request, headers=headers)
    # Usually returns 200 even with write errors
    print(response.status_code)
    if response.status_code == 200:
        result = json.loads(response.content)
        status = result.get("status", {})
        # real code, if the request is successful, this field will not be returned
        code = status.get("code", 0)
        # If the request is successful, the message value is 'OK', otherwise it will be a failure message
        message = status.get("message", "")
        print(code)
        print(message)
        if code != 0 and code != 200:
            errors = result.get("errors", [])
            for err in errors:
                print(f'failure message: {err.get("message")}, failure sample data: {err.get("data")}')
```
