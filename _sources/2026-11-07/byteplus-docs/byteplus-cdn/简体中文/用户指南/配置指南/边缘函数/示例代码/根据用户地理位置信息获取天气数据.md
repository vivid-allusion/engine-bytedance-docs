该示例展示了如何根据用户的地理位置信息获取天气数据。

您通过 [FetchEvent 接口](https://docs.byteplus.com/en/docs/byteplus-cdn/docs_cn-FetchEvent-cn)的 `info.geo` 属性获取用户的地理位置信息。示例中假设您通过 https://www.tianqiapi.com 提供的 API 获取天气数据。您需要在请求中传入查询参数 city 指定用户所在的城市。

```JavaScript
addEventListener('fetch', (event) => {
  event.respondWith(handle(event));
});

//您需要将apiURL中appsecret和appid的内容更改为真实信息
const apiURL =
    'https://www.tianqiapi.com/free/day?appsecret=xxxxxx&appid=xxxxxx&';

async function queryWeather(city) {
  const response = await fetch(apiURL + 'city=' + city);
  if (response.ok) {
    // 假设返回的数据为JSON
    /*
    {
      air: '94', city: '南京', cityid: '101190101', tem: '31', tem_day: '31',
          tem_night: '24', update_time: '14:12', wea: '多云', wea_img: 'yun',
          win: '东南风', win_meter: '9km/h', win_speed: '2级'
    }
    */

    const data = await response.json();

    const weather = data.wea;
    const tempDay = data.tem_day;
    const tempNight = data.tem_night;


    return `
     来自${city} 的用户
     您好，您的当地天气为：${weather}，白天气温为 ${
        tempDay}，夜间气温为 ${tempNight}
    
    `;
  } else {
    return '天气API开小差了';
  }
}

async function handle(event) {
  // 获得客户端的ip城市地址信息
  const city = event.info.geo.city;
  // 通过city查询该用户的天气信息
  return queryWeather(city);
}


```
