本示例展示如何根據使用者的地理位置資訊取得天氣資料。示例中假設您透過 https://www.tianqiapi.com 取得天氣資料。您需要在請求中傳入查詢參數 city 指定使用者所在的城市。

```JavaScript
addEventListener('fetch', (event) => {
  event.respondWith(handle(event));
});

// 您需要將apiURL中appsecret和appid的內容更改為真實資訊
const apiURL =
    'https://www.tianqiapi.com/free/day?appsecret=xxxxxx&appid=xxxxxx&';

async function queryWeather(city) {
  const response = await fetch(apiURL + 'city=' + city);
  if (response.ok) {
    // 假設返回的資料為JSON
    /*
    {
      air: '94', city: '南京', cityid: '101190101', tem: '31', tem_day: '31',
          tem_night: '24', update_time: '14:12', wea: '多雲', wea_img: 'yun',
          win: '東南風', win_meter: '9km/h', win_speed: '2級'
    }
    */

    const data = await response.json();

    const weather = data.wea;
    const tempDay = data.tem_day;  // 修正變數名
    const tempNight = data.tem_night;  // 修正變數名

    return `
     來自${city} 的使用者
     您好，您的當地天氣為：${weather}，白天氣溫為 ${tempDay}，夜間氣溫為 ${tempNight}
    `;
  } else {
    return '天氣API暫時無法使用';
  }
}

async function handle(event) {
  // 取得用戶端的IP城市資訊
  const city = event.info.geo.city;
  // 透過city查詢該使用者的天氣資訊
  return new Response(await queryWeather(city), {
    headers: {'Content-Type': 'text/html; charset=UTF-8'}
  });
}
```
